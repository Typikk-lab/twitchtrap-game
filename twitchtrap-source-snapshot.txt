import { useEffect, useRef, useState } from 'react'
import './ArcadeGame.css'
import { createAudioEngine } from './audio.js'
import {
  GAME_HEIGHT as HEIGHT,
  GAME_WIDTH as WIDTH,
  getTargetAtElapsed,
  isCursorInsideTarget,
} from './arcadeRules.js'

const COUNTDOWN_SECONDS = 3

const DEGEN_TAUNTS = [
  { title: 'BRO GOT LIQUIDATED', subtitle: 'The target was literally right there.' },
  { title: 'SKILL ISSUE DETECTED', subtitle: 'Your hands have paper hands.' },
  { title: 'ABSOLUTE CINEMA', subtitle: 'Thirty seconds of confidence. One frame of consequences.' },
  { title: 'NGMI.', subtitle: 'The trap remains undefeated. Your comeback is pending.' },
  { title: 'EXIT LIQUIDITY FOUND', subtitle: 'You were the liquidity.' },
  { title: 'WITNESSED. RECORDED. MOCKED.', subtitle: 'The replay is going straight to the group chat.' },
  { title: 'TOUCH GRASS? YOU TOUCHED DEATH.', subtitle: 'The box said stay inside. You chose freedom.' },
  { title: 'BRO FUMBLED THE BAG', subtitle: 'A generational fumble in real time.' },
  { title: 'NPC MOVEMENT DETECTED', subtitle: 'Even the tutorial is disappointed.' },
  { title: 'PACK IT UP', subtitle: 'The run is over. The screenshots are forever.' },
]

export default function ArcadeGame() {
  const canvasRef = useRef(null)
  const audioRef = useRef(null)
  const [isMuted, setIsMuted] = useState(() => {
    try {
      return window.localStorage.getItem('twitchtrap-muted') === 'true'
    } catch {
      return false
    }
  })
  const modeRef = useRef('IDLE')
  const cursorRef = useRef({ x: -100, y: -100 })
  const startRef = useRef(0)
  const lastHudSecondRef = useRef(-1)
  const boxRef = useRef({ x: 345, y: 155, size: 110 })

  const [mode, setMode] = useState('IDLE')
  const [elapsed, setElapsed] = useState(0)
  const [multiplier, setMultiplier] = useState(1)
  const [result, setResult] = useState(null)
  const [verificationStatus, setVerificationStatus] = useState('local')
  const [bestTime, setBestTime] = useState(() => {
    try {
      return Number(window.localStorage.getItem('twitchtrap-best-time')) || 0
    } catch {
      return 0
    }
  })

  useEffect(() => {
    audioRef.current = createAudioEngine()
    audioRef.current.setMuted(isMuted)

    return () => {
      audioRef.current?.dispose()
      audioRef.current = null
    }
  }, [])

  useEffect(() => {
    audioRef.current?.setMuted(isMuted)
  }, [isMuted])

  useEffect(() => {
    if (mode === 'DEAD') {
      audioRef.current?.setScene('death')
    } else if (mode === 'ACTIVE') {
      audioRef.current?.setScene('gameplay')
    } else {
      audioRef.current?.setScene('menu')
    }
  }, [mode])

  useEffect(() => {
    audioRef.current?.setIntensity(Math.min(elapsed / 30, 1))
  }, [elapsed])
  useEffect(() => {
    const canvas = canvasRef.current
    const ctx = canvas?.getContext('2d')
    if (!canvas || !ctx) return undefined

    let animationId = 0

    function drawGrid() {
      ctx.fillStyle = '#080812'
      ctx.fillRect(0, 0, WIDTH, HEIGHT)

      ctx.strokeStyle = 'rgba(130, 95, 210, 0.16)'
      ctx.lineWidth = 1

      for (let x = 0; x <= WIDTH; x += 24) {
        ctx.beginPath()
        ctx.moveTo(x, 0)
        ctx.lineTo(x, HEIGHT)
        ctx.stroke()
      }

      for (let y = 0; y <= HEIGHT; y += 24) {
        ctx.beginPath()
        ctx.moveTo(0, y)
        ctx.lineTo(WIDTH, y)
        ctx.stroke()
      }

      const glow = ctx.createRadialGradient(
        WIDTH / 2, HEIGHT / 2, 20,
        WIDTH / 2, HEIGHT / 2, 330,
      )
      glow.addColorStop(0, 'rgba(95, 40, 170, 0.12)')
      glow.addColorStop(1, 'rgba(5, 5, 12, 0)')
      ctx.fillStyle = glow
      ctx.fillRect(0, 0, WIDTH, HEIGHT)
    }

    function drawTarget(box, color, lineWidth = 2) {
      ctx.save()
      ctx.strokeStyle = color
      ctx.lineWidth = lineWidth
      ctx.shadowColor = color
      ctx.shadowBlur = 14
      ctx.strokeRect(box.x, box.y, box.size, box.size)
      ctx.shadowBlur = 0

      const cx = box.x + box.size / 2
      const cy = box.y + box.size / 2

      ctx.lineWidth = 1
      ctx.beginPath()
      ctx.arc(cx, cy, 10, 0, Math.PI * 2)
      ctx.stroke()

      ctx.beginPath()
      ctx.moveTo(cx - 17, cy)
      ctx.lineTo(cx + 17, cy)
      ctx.moveTo(cx, cy - 17)
      ctx.lineTo(cx, cy + 17)
      ctx.stroke()
      ctx.restore()
    }

    function drawCursor() {
      const cursor = cursorRef.current
      if (cursor.x < 0 || cursor.y < 0) return

      ctx.save()
      ctx.fillStyle = '#ff3caa'
      ctx.shadowColor = '#ff3caa'
      ctx.shadowBlur = 12
      ctx.beginPath()
      ctx.arc(cursor.x, cursor.y, 4, 0, Math.PI * 2)
      ctx.fill()
      ctx.restore()
    }

    function drawLabel(text, y, color = '#f5f0ff', size = 13) {
      ctx.save()
      ctx.fillStyle = color
      ctx.font = `900 ${size}px ui-monospace, monospace`
      ctx.textAlign = 'center'
      ctx.fillText(text, WIDTH / 2, y)
      ctx.restore()
    }

    function finishRun(title, subtitle, time) {
      audioRef.current?.death()
      modeRef.current = 'DEAD'
      setMode('DEAD')
      const taunt = DEGEN_TAUNTS[Math.floor(Math.random() * DEGEN_TAUNTS.length)]
      const resultTitle = time <= 0 ? title : taunt.title
      const resultSubtitle = time <= 0 ? subtitle : taunt.subtitle
      setResult({ title: resultTitle, subtitle: resultSubtitle, time })

      if (time > 0) {
        setBestTime((previousBest) => {
          if (time <= previousBest) return previousBest

          try {
            window.localStorage.setItem('twitchtrap-best-time', String(time))
          } catch {
            // The game remains playable if browser storage is unavailable.
          }

          return time
        })
      }
    }

    function beginActiveRun(timestamp) {
      audioRef.current?.lockIn()
      modeRef.current = 'ACTIVE'
      startRef.current = timestamp
      lastHudSecondRef.current = -1
      setMode('ACTIVE')
    }

    function frame(timestamp) {
      drawGrid()

      const currentMode = modeRef.current
      let box = boxRef.current

      if (currentMode === 'IDLE') {
        box = { x: WIDTH / 2 - 55, y: HEIGHT / 2 - 55, size: 110 }
        boxRef.current = box
        drawTarget(box, '#9b72ff')
        drawLabel('READY WHEN YOU ARE', 42, '#d6ff3f')
        drawLabel('KEEP YOUR CURSOR INSIDE THE BOX', HEIGHT - 25, '#a5a2b5', 11)
      }

      if (currentMode === 'PREPARING') {
        box = { x: WIDTH / 2 - 55, y: HEIGHT / 2 - 55, size: 110 }
        boxRef.current = box
        drawTarget(box, '#9b72ff')
        drawLabel('PREPARING RUN...', 42, '#d6ff3f')
        drawLabel('REQUESTING RUN CHALLENGE', HEIGHT - 25, '#a5a2b5', 11)
        drawCursor()
      }

      if (currentMode === 'COUNTDOWN') {
        const elapsedCountdown = (timestamp - startRef.current) / 1000
        const remaining = Math.max(0, COUNTDOWN_SECONDS - elapsedCountdown)

        box = { x: WIDTH / 2 - 55, y: HEIGHT / 2 - 55, size: 110 }
        boxRef.current = box

        const locked = isCursorInsideTarget(cursorRef.current, box)
        drawTarget(box, locked ? '#00ff66' : '#ffcc00', 3)
        drawLabel(`GET READY: ${remaining.toFixed(1)}s`, 42)
        drawLabel(
          locked ? 'TARGET LOCKED' : 'MOVE YOUR CURSOR INTO THE BOX',
          HEIGHT - 25,
          locked ? '#00ff66' : '#ffcc00',
          11,
        )
        drawCursor()

        if (remaining <= 0) {
          if (!locked) {
            finishRun(
              'ELIMINATED',
              'You left the target before the run began.',
              0,
            )
          } else {
            beginActiveRun(timestamp)
          }
        }
      }

      if (currentMode === 'ACTIVE' || modeRef.current === 'ACTIVE') {
        const activeElapsed = Math.max(0, (timestamp - startRef.current) / 1000)
        box = getTargetAtElapsed(activeElapsed)
        const size = box.size
        boxRef.current = box

        drawTarget(box, '#00eaff')
        drawCursor()

        const cursor = cursorRef.current
        const outOfBounds = !isCursorInsideTarget(cursor, box)

        drawLabel(`SURVIVED ${activeElapsed.toFixed(2)}s`, 34, '#ffffff', 12)
        drawLabel(
          `TARGET SIZE ${Math.round(size)}px`,
          HEIGHT - 20,
          '#00eaff',
          10,
        )

        const wholeSecond = Math.floor(activeElapsed)
        if (wholeSecond !== lastHudSecondRef.current) {
          lastHudSecondRef.current = wholeSecond
          setElapsed(activeElapsed)
          setMultiplier(1 + activeElapsed * 0.1)
          audioRef.current?.runPulse(wholeSecond)
        }

        if (outOfBounds) {
          finishRun('LIQUIDATED!', 'The target escaped you. Try again.', activeElapsed)
        } else {
          animationId = requestAnimationFrame(frame)
          return
        }
      }

      if (modeRef.current === 'DEAD') {
        drawTarget(boxRef.current, '#ff3caa')
        drawCursor()
      }

      animationId = requestAnimationFrame(frame)
    }

    animationId = requestAnimationFrame(frame)

    return () => cancelAnimationFrame(animationId)
  }, [])

  function toggleMute() {
    if (!audioRef.current?.isReady()) {
      if (isMuted) {
        setIsMuted(false)
        try {
          window.localStorage.setItem('twitchtrap-muted', 'false')
        } catch {
          // Audio remains available for this session.
        }
      }
      void audioRef.current?.unlock()
      return
    }

    setIsMuted((previous) => {
      const next = !previous
      try {
        window.localStorage.setItem('twitchtrap-muted', String(next))
      } catch {
        // Mute still works for this session if storage is unavailable.
      }
      return next
    })
  }

  function updateCursor(event) {
    const canvas = canvasRef.current
    if (!canvas) return

    const rect = canvas.getBoundingClientRect()
    cursorRef.current = {
      x: ((event.clientX - rect.left) / rect.width) * WIDTH,
      y: ((event.clientY - rect.top) / rect.height) * HEIGHT,
    }
  }

    if (['PREPARING', 'COUNTDOWN', 'ACTIVE'].includes(modeRef.current)) return

    audioRef.current?.unlock()
    setElapsed(0)
    setMultiplier(1)
    setResult(null)

    lastHudSecondRef.current = -1
    modeRef.current = 'COUNTDOWN'
    setMode('COUNTDOWN')
  }

  const running =
    mode === 'PREPARING' || mode === 'COUNTDOWN' || mode === 'ACTIVE'

  return (

                : mode === 'DEAD'

export const GAME_WIDTH = 800
export const GAME_HEIGHT = 420
export const MAX_RAMP_SECONDS = 30

export function getTargetAtElapsed(elapsedSeconds) {
  if (!Number.isFinite(elapsedSeconds)) {
    throw new TypeError('Elapsed time must be a finite number')
  }

  const elapsed = Math.max(0, elapsedSeconds)
  const ramp = Math.min(elapsed / MAX_RAMP_SECONDS, 1)
  const size = Math.max(45, 110 - ramp * 65)
  const speed = 0.6 + ramp * 2.1
  const ampX = 70 + ramp * 80
  const ampY = 40 + ramp * 50
  const centerX = GAME_WIDTH / 2 - size / 2
  const centerY = GAME_HEIGHT / 2 - size / 2

  return {
    x: centerX + Math.sin(elapsed * speed) * ampX,
    y: centerY + Math.sin(elapsed * speed * 1.3) * ampY,
    size,
  }
}

export function isCursorInsideTarget(cursor, target) {
  if (
    !cursor ||
    !target ||
    !Number.isFinite(cursor.x) ||
    !Number.isFinite(cursor.y) ||
    !Number.isFinite(target.x) ||
    !Number.isFinite(target.y) ||
    !Number.isFinite(target.size) ||
    target.size < 0
  ) {
    return false
  }

  return (
    cursor.x >= target.x &&
    cursor.x <= target.x + target.size &&
    cursor.y >= target.y &&
    cursor.y <= target.y + target.size
  )
}

export function createAudioEngine() {
  let context = null
  let master = null
  let muted = false
  let scene = 'menu'
  let intensity = 0
  let step = 0
  let timer = null
  let deathTimer = null
  let lastPulseSecond = -1

  const menuNotes = [659, 0, 523, 784, 0, 587, 392, 0, 523, 0, 880, 784, 0, 587, 494, 0]
  const gameNotes = [110, 165, 220, 330, 247, 370, 294, 440, 330, 494, 370, 554, 330, 494, 247, 370]

  function ensureContext() {
    if (typeof window === 'undefined') return null

    const AudioContextClass = window.AudioContext || window.webkitAudioContext
    if (!AudioContextClass) return null

    if (!context) {
      context = new AudioContextClass()
      master = context.createGain()
      master.gain.value = muted ? 0 : 0.65
      master.connect(context.destination)
    }

    return context
  }

  async function unlock() {
    const audio = ensureContext()
    if (!audio) return false

    if (audio.state === 'suspended') {
      try {
        await audio.resume()
      } catch {
        return false
      }
    }

    startLoop()
    return audio.state === 'running'
  }

  function tone(frequency, duration, options = {}) {
    if (muted || !context || !master || context.state !== 'running') return

    const {
      type = 'sine',
      endFrequency = frequency,
      volume = 0.12,
      delay = 0,
    } = options

    const start = context.currentTime + delay
    const oscillator = context.createOscillator()
    const envelope = context.createGain()

    oscillator.type = type
    oscillator.frequency.setValueAtTime(Math.max(20, frequency), start)
    oscillator.frequency.exponentialRampToValueAtTime(
      Math.max(20, endFrequency),
      start + duration,
    )

    envelope.gain.setValueAtTime(0.0001, start)
    envelope.gain.exponentialRampToValueAtTime(Math.max(0.0002, volume), start + 0.012)
    envelope.gain.exponentialRampToValueAtTime(0.0001, start + duration)

    oscillator.connect(envelope)
    envelope.connect(master)
    oscillator.start(start)
    oscillator.stop(start + duration + 0.02)
  }

  function tick() {
    if (muted || !context || context.state !== 'running') return

    if (scene === 'menu') {
      const note = menuNotes[step % menuNotes.length]

      if (note > 0) {
        tone(note, 0.105, {
          type: 'square',
          endFrequency: note * 0.985,
          volume: 0.045,
        })
      }

      if (step % 4 === 0) {
        tone(step % 8 === 0 ? 98 : 123, 0.16, {
          type: 'triangle',
          endFrequency: 70,
          volume: 0.075,
        })
      }

      if (step % 8 === 6) {
        tone(1047, 0.07, {
          type: 'square',
          endFrequency: 784,
          volume: 0.025,
          delay: 0.035,
        })
      }

      step += 1
      return
    }

    if (scene === 'gameplay') {
      const note = gameNotes[step % gameNotes.length]
      const pitch = 1 + intensity * 0.45
      const volume = 0.045 + intensity * 0.025

      tone(note * pitch, 0.095, {
        type: 'square',
        endFrequency: note * pitch * 1.015,
        volume,
      })

      if (step % 4 === 0) {
        tone(55 + intensity * 22, 0.19, {
          type: 'sawtooth',
          endFrequency: 38 + intensity * 10,
          volume: 0.075 + intensity * 0.025,
        })
      }

      if (step % 8 === 4) {
        tone(660 + intensity * 440, 0.065, {
          type: 'triangle',
          endFrequency: 880 + intensity * 550,
          volume: 0.035 + intensity * 0.025,
          delay: 0.035,
        })
      }

      step += 1
    }
  }

  function startLoop() {
    if (timer !== null || muted || !context || context.state !== 'running') return
    timer = window.setInterval(tick, 125)
  }

  function stopLoop() {
    if (timer !== null) {
      window.clearInterval(timer)
      timer = null
    }
  }

  function setScene(nextScene) {
    if (scene === nextScene) return
    scene = nextScene
    step = 0
    stopLoop()
    startLoop()
  }

  function setIntensity(nextIntensity) {
    intensity = Math.max(0, Math.min(1, nextIntensity))
  }

  function isReady() {
    return Boolean(context && context.state === 'running')
  }

  function lockIn() {
    tone(440, 0.16, { type: 'triangle', endFrequency: 660, volume: 0.14 })
    tone(880, 0.22, {
      type: 'sine',
      endFrequency: 1320,
      volume: 0.09,
      delay: 0.11,
    })
  }

  function runPulse(second) {
    if (second <= 0 || second % 3 !== 0 || second === lastPulseSecond) return

    lastPulseSecond = second
    const urgency = Math.min(second / 30, 1)

    tone(180 + urgency * 100, 0.12, {
      type: 'triangle',
      endFrequency: 100 + urgency * 40,
      volume: 0.035 + urgency * 0.025,
    })
  }

  function death() {
    if (deathTimer !== null) {
      window.clearTimeout(deathTimer)
      deathTimer = null
    }

    setScene('death')

    // A short, exaggerated arcade elimination sting.
    tone(115, 0.48, { type: 'sawtooth', endFrequency: 32, volume: 0.24 })
    tone(68, 0.38, { type: 'square', endFrequency: 35, volume: 0.12, delay: 0.025 })
    tone(920, 0.07, { type: 'square', endFrequency: 170, volume: 0.055, delay: 0.04 })
    tone(440, 0.09, { type: 'sawtooth', endFrequency: 90, volume: 0.045, delay: 0.13 })

    // Return to the menu soundtrack after the sting finishes.
    deathTimer = window.setTimeout(() => {
      deathTimer = null
      setScene('menu')
    }, 680)
  }

  function setMuted(nextMuted) {
    muted = Boolean(nextMuted)

    if (master && context) {
      master.gain.setTargetAtTime(
        muted ? 0 : 0.65,
        context.currentTime,
        0.025,
      )
    }

    if (muted) stopLoop()
    else startLoop()
  }

  function dispose() {
    stopLoop()
    if (deathTimer !== null) {
      window.clearTimeout(deathTimer)
      deathTimer = null
    }

    if (context && context.state !== 'closed') {
      context.close().catch(() => {})
    }

    context = null
    master = null
  }

  return {
    unlock,
    lockIn,
    runPulse,
    death,
    setMuted,
    setScene,
    setIntensity,
    isReady,
    dispose,
  }
}

import { createClient } from "npm:@supabase/supabase-js@2.57.4";

const corsHeaders = {
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
  "Content-Type": "application/json",
};

function allowedOrigins(): Set<string> {
  const configured = Deno.env.get("ARCADE_ALLOWED_ORIGINS");
  return new Set(
    (configured
      ? configured.split(",")
      : ["http://localhost:5173", "http://127.0.0.1:5173"])
      .map((origin) => origin.trim())
      .filter(Boolean),
  );
}

function respond(
  req: Request,
  body: Record<string, unknown>,
  status = 200,
): Response {
  const origin = req.headers.get("origin");
  const headers = new Headers(corsHeaders);

  if (origin && allowedOrigins().has(origin)) {
    headers.set("Access-Control-Allow-Origin", origin);
    headers.set("Vary", "Origin");
  }

  return new Response(JSON.stringify(body), { status, headers });
}

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") {
    const origin = req.headers.get("origin");
    if (origin && !allowedOrigins().has(origin)) {
      return respond(req, { error: "Origin not allowed" }, 403);
    }
    return new Response(null, {
      status: 204,
      headers: {
        ...corsHeaders,
        ...(origin ? { "Access-Control-Allow-Origin": origin, "Vary": "Origin" } : {}),
      },
    });
  }

  if (req.method !== "POST") {
    return respond(req, { error: "Method not allowed" }, 405);
  }

  const origin = req.headers.get("origin");
  if (origin && !allowedOrigins().has(origin)) {
    return respond(req, { error: "Origin not allowed" }, 403);
  }

  let body: unknown;
  try {
    body = await req.json();
  } catch {
    return respond(req, { error: "Request body must be valid JSON" }, 400);
  }

  const challengeId =
    typeof body === "object" && body !== null && "challenge_id" in body
      ? (body as Record<string, unknown>).challenge_id
      : null;

  if (
    typeof challengeId !== "string" ||
    !/^[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(challengeId)
  ) {
    return respond(req, { error: "A valid challenge_id is required" }, 400);
  }

  const url = Deno.env.get("SUPABASE_URL");
  const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");

  if (!url || !serviceKey) {
    console.error("Missing server-side Supabase environment configuration");
    return respond(req, { error: "Server configuration unavailable" }, 500);
  }

  try {
    const supabase = createClient(url, serviceKey, {
      auth: { persistSession: false, autoRefreshToken: false },
    });

    const { data, error } = await supabase.rpc("activate_arcade_challenge", {
      p_challenge_id: challengeId,
    });

    if (error) {
      console.error("Challenge activation RPC failed:", error.message);
      return respond(req, { error: "Could not activate challenge" }, 400);
    }

    const challenge = Array.isArray(data) ? data[0] : null;
    if (
      !challenge?.challenge_id ||
      !challenge.activated_at ||
      !challenge.expires_at ||
      !challenge.game_rules_version
    ) {
      console.error("Challenge activation returned an unexpected result");
      return respond(req, { error: "Invalid activation response" }, 500);
    }

    return respond(req, { challenge });
  } catch (error) {
    console.error("Challenge activation failed:", error);
    return respond(req, { error: "Internal server error" }, 500);
  }
});

  status = 200,
): Response {
  const origin = req.headers.get("origin");
  const headers = new Headers(corsHeaders);

  if (origin && allowedOrigins().has(origin)) {
    headers.set("Access-Control-Allow-Origin", origin);
    headers.set("Vary", "Origin");
  }

  return new Response(JSON.stringify(body), { status, headers });
}

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") {
    const origin = req.headers.get("origin");
    if (origin && !allowedOrigins().has(origin)) {
      return respond(req, { error: "Origin not allowed" }, 403);
    }
    return new Response(null, {
      status: 204,
      headers: {
        ...corsHeaders,
        ...(origin ? { "Access-Control-Allow-Origin": origin, "Vary": "Origin" } : {}),
      },
    });
  }

  if (req.method !== "POST") {
    return respond(req, { error: "Method not allowed" }, 405);
  }

  const origin = req.headers.get("origin");
  if (origin && !allowedOrigins().has(origin)) {
    return respond(req, { error: "Origin not allowed" }, 403);
  }

  const url = Deno.env.get("SUPABASE_URL");
  const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");

  if (!url || !serviceKey) {
    console.error("Missing server-side Supabase environment configuration");
    return respond(req, { error: "Server configuration unavailable" }, 500);
  }

  try {
    const supabase = createClient(url, serviceKey, {
      auth: { persistSession: false, autoRefreshToken: false },
    });

    const { data, error } = await supabase.rpc("issue_arcade_challenge");

    if (error) {
      console.error("Challenge issuance RPC failed:", error.message);
      return respond(req, { error: "Could not issue challenge" }, 500);
    }

    const challenge = Array.isArray(data) ? data[0] : null;
    if (
      !challenge?.challenge_id ||
      challenge.seed === undefined ||
      !challenge.expires_at ||
      !challenge.game_rules_version
    ) {
      console.error("Challenge issuance returned an unexpected result");
      return respond(req, { error: "Invalid challenge response" }, 500);
    }

    return respond(req, { challenge });
  } catch (error) {
    console.error("Challenge issuance failed:", error);
    return respond(req, { error: "Internal server error" }, 500);
  }
});
